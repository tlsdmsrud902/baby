```js
// pet → baby앙 일괄 치환
const fs = require('fs');
const { execSync } = require('child_process');
const files = execSync('git -c core.quotepath=off ls-files', { encoding: 'utf8' }).split('\n').filter(Boolean)
  .filter(f => /\.(html|css|js|json|txt|xml|md|cjs)$/i.test(f) && fs.existsSync(f));

const img = { // 옛 이미지 이름 → 새 이미지 이름 (4-3 표)

const names = Object.keys(img).sort((a, b) => b.length - a.length).join('|');

const rules = [

  [new RegExp('(?<![a-z-])(' + names + ')\\.(png|webp|mp4)', 'g'), (m, n) => img[n] + '.webp'],

  [/(?<![A-Za-z])pets(?![a-z])/g, 'kids'], [/(?<![A-Za-z])pet(?![a-z])/g, 'baby'],
```

### 4-2. 변환 명령 (ffmpeg → webp)

```bash
O=…/skin1/SkinImg/baby
# 가로 장면 1672px
ffmpeg -y -i 원본.jpg -vf "scale=1672:-2" -c:v libwebp -quality 80 $O/scene-park.webp
# 정사각 1254px (0.5 = 가운데, 0.3 = 왼쪽으로)
ffmpeg -y -i 원본.jpg -vf "crop=ih:ih:(iw-ih)*0.5:0,scale=1254:1254" -c:v libwebp -quality 80 $O/sq-sleep.webp
# 세로 3:4 카드 1086×1448
ffmpeg -y -i 원본.png -vf "crop=ih*3/4:ih:(iw-ih*3/4)/2:0,scale=1086:1448" -c:v libwebp -quality 80 $O/card-newborn.webp
# 세로가 더 긴 사진의 정사각 : crop=iw:iw:0:(ih-iw)*0.3
```

**로고 (글자 로고, 배경 투명)** — 맑은 고딕 굵게

```bash
F="C\\:/Windows/Fonts/malgunbd.ttf"
ffmpeg -y -f lavfi -i "color=c=black@0.0:s=560x200,format=rgba" -frames:v 1 \
  -vf "drawtext=fontfile='$F':text='baby앙':fontsize=132:fontcolor=0x43372f:x=(w-text_w)/2:y=(h-text_h)/2-8" \
```

```json
{ "code": "p02", "group": "rec", "cates": [24, 27], "name": "순면 배냇저고리 세트", "price": 28000, "retail": 35000, "img": "p02.jpg" }
```

상품 이미지는 받은 사진에서 정사각형으로 잘라 만들었다. (중심 비율 cx·cy, 크기 s = 짧은 변 대비)

```js
// 사용: node crops.js '[["p01","A",0.45,0.55,0.9], ["p05","A",0.77,0.2,0.36], …]'
const { execFileSync } = require('child_process');
const S = { A: 'Baby_sleeping_on_linen_blanket_2K_….jpg' /* 사진 별칭 : 파일명 */ };
const dims = f => execFileSync('ffprobe', ['-v','error','-show_entries','stream=width,height','-of','csv=p=0', f], { encoding: 'utf8' }).trim().split(',').map(Number);
for (const [code, src, cx, cy, s] of JSON.parse(process.argv[2])) {
  const f = S[src], [w, h] = dims(f), side = Math.round(Math.min(w, h) * s);
  const x = Math.max(0, Math.min(w - side, Math.round(w * cx - side / 2)));
  const y = Math.max(0, Math.min(h - side, Math.round(h * cy - side / 2)));
  execFileSync('ffmpeg', ['-v','error','-y','-i',f,'-vf',`crop=${side}:${side}:${x}:${y},scale=800:800`,'-q:v','3',`cafe24-assets/products/${code}.jpg`]);
}
```

```js
// 정적 파일 : root 기준 경로가 root 안인지 확인 (root 는 반드시 path.resolve 로 정규화 — 안 하면 전부 403)
```

## 10. 이번 작업(baby앙)에서 겪은 문제와 해결

| 문제 | 원인 | 해결 |
|---|---|---|
| 로컬 미리보기에서 CSS · 이미지가 전부 403 | 서버 `root` 경로가 `D:/…` 형식이라 `path.resolve` 결과(`D:\…`)와 비교가 안 맞음 | `root` 를 `path.resolve()` 로 감싸기 |
| 메뉴에 신생아 · SALE 이 안 보임 | 새 계정에 상품분류가 없는데 관리자 분류로 메뉴를 다시 그림 | `menu.editorial: true` |
| `git mv` 가 Permission denied | 터미널 · 미리보기 서버가 폴더를 잡고 있음 | 서버 중지, 작업 위치를 상위 폴더로 옮긴 뒤 재시도 |
| 셸에서 node 스크립트를 넘길 때 `\s`, `\d` 가 사라짐 | 셸을 거치며 백슬래시가 빠짐 | 정규식이 들어간 코드는 파일로 저장해 직접 편집 |
| 폴더에서 지웠는데 GitHub 기록에 옛 파일이 남음 | 예전 커밋에 그대로 있음 | orphan 브랜치로 새 기록 만들고 `push -f` |
| 조사 틀림 (`baby앙가`) | 브랜드 받침이 바뀜 | 3-3 검사 |
